--SQL Queries
-- Find the transactions between 4pm and 7pm everyday grouped by scanner
SELECT 
    t.device_id,
    s.location,
    COUNT(t.txn_id) AS total_transactions,
    SUM(t.amount) AS total_amount
FROM transactions t
JOIN scanners s ON t.device_id = s.device_id
WHERE EXTRACT(HOUR FROM t.txn_time) BETWEEN 16 AND 18
GROUP BY t.device_id, s.location
ORDER BY total_transactions DESC;


-- Display the users with balance lower than ₹900
SELECT 
    u.roll_no,
    u.name,
    w.balance
FROM users u
JOIN wallets w ON u.roll_no = w.roll_no
WHERE w.balance < 900
ORDER BY w.balance ASC;


--PL/SQL Functions and Procedures and their executions
-- PL/SQL Function that will return the balance of the asked roll_no
CREATE OR REPLACE FUNCTION get_balance(p_roll_no IN NUMBER)
RETURN NUMBER
IS
    v_balance NUMBER;
BEGIN
    SELECT balance
    INTO v_balance
    FROM wallets
    WHERE roll_no = p_roll_no;

    RETURN v_balance;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN NULL;
END;

-- PL/SQL Procedure to update wallet active status
CREATE OR REPLACE PROCEDURE update_wallet_active_status(
    p_roll_no   IN NUMBER,
    p_is_active IN NUMBER
)
IS
    v_wallet_id NUMBER;
BEGIN
    -- Validate input
    IF p_is_active NOT IN (0, 1) THEN
        DBMS_OUTPUT.PUT_LINE('Warning: is_active must be 0 or 1. Skipping update.');
        RETURN;
    END IF;

    -- Check if wallet exists
    BEGIN
        SELECT wallet_id
        INTO v_wallet_id
        FROM wallets
        WHERE roll_no = p_roll_no;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for roll number ' || p_roll_no || '. Skipping update.');
            RETURN;
    END;

    -- Update the is_active flag
    UPDATE wallets
    SET is_active   = p_is_active,
        last_updated = SYSTIMESTAMP
    WHERE roll_no = p_roll_no;

    -- Commit the transaction
    COMMIT;

    DBMS_OUTPUT.PUT_LINE('Wallet active status updated for roll number ' || p_roll_no ||
                         ' (is_active: ' || p_is_active || ').');

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error updating wallet active status for roll number ' || p_roll_no || ': ' || SQLERRM);
END;

-- PL/SQL Procedure to process wallet balance topups
CREATE OR REPLACE PROCEDURE process_topup(
    p_roll_no IN NUMBER,
    p_amount IN NUMBER,
    p_payment_method IN VARCHAR2,
    p_ref_txn_id IN VARCHAR2
)
IS
    v_wallet_id NUMBER;
    v_current_balance NUMBER;
    v_next_topup_id NUMBER;
    v_updated_balance NUMBER;
BEGIN
    -- Validate inputs
    IF p_amount <= 0 THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Top-up amount must be positive. Skipping top-up.');
        RETURN;
    END IF;
    IF p_ref_txn_id IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Reference transaction ID cannot be null. Skipping top-up.');
        RETURN;
    END IF;

    -- Check if wallet is active
    DECLARE
        v_is_active NUMBER;
    BEGIN
        SELECT is_active
        INTO v_is_active
        FROM wallets
        WHERE roll_no = p_roll_no;
        IF v_is_active = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Warning: Wallet for roll number ' || p_roll_no || ' is inactive. Skipping top-up.');
            RETURN;
        END IF;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for roll number ' || p_roll_no || '. Skipping top-up.');
            RETURN;
    END;

    -- Get wallet details
    BEGIN
        SELECT wallet_id, balance
        INTO v_wallet_id, v_current_balance
        FROM wallets
        WHERE roll_no = p_roll_no;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for roll number ' || p_roll_no || '. Skipping top-up.');
            RETURN;
    END;

    -- Determine next topup_id
    SELECT NVL(MAX(topup_id), 0) + 1
    INTO v_next_topup_id
    FROM topups;

    -- Insert top-up record
    INSERT INTO topups (topup_id, roll_no, amount, payment_method, ref_txn_id, topup_time)
    VALUES (v_next_topup_id, p_roll_no, p_amount, p_payment_method, p_ref_txn_id, SYSTIMESTAMP);

    -- Update wallet balance
    UPDATE wallets
    SET balance = v_current_balance + p_amount,
        transaction_counter = transaction_counter + 1,
        last_updated = SYSTIMESTAMP
    WHERE wallet_id = v_wallet_id;

    -- Commit the transaction
    COMMIT;

    -- Fetch the updated balance
    SELECT balance
    INTO v_updated_balance
    FROM wallets
    WHERE roll_no = p_roll_no;

    -- Output the success message with updated balance
    DBMS_OUTPUT.PUT_LINE('Top-up successful for roll number ' || p_roll_no ||
                         ' (new balance: ' || v_updated_balance || ').');

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error during top-up for roll number ' || p_roll_no || ': ' || SQLERRM);
END;

-- PL/SQL Procedure to process transactions
CREATE OR REPLACE PROCEDURE process_transaction(
    p_roll_no IN NUMBER,
    p_amount IN NUMBER,
    p_device_id IN VARCHAR2
)
IS
    v_wallet_id NUMBER;
    v_current_balance NUMBER;
    v_next_txn_id NUMBER;
    v_updated_balance NUMBER;
BEGIN
    -- Validate inputs
    IF p_amount <= 0 THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Transaction amount must be positive. Skipping transaction.');
        RETURN;
    END IF;
    IF p_device_id IS NULL THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Device ID cannot be null. Skipping transaction.');
        RETURN;
    END IF;

    -- Check if wallet is active
    DECLARE
        v_is_active NUMBER;
    BEGIN
        SELECT is_active
        INTO v_is_active
        FROM wallets
        WHERE roll_no = p_roll_no;
        IF v_is_active = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Warning: Wallet for roll number ' || p_roll_no || ' is inactive. Skipping transaction.');
            RETURN;
        END IF;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for roll number ' || p_roll_no || '. Skipping transaction.');
            RETURN;
    END;

    -- Get wallet details
    BEGIN
        SELECT wallet_id, balance
        INTO v_wallet_id, v_current_balance
        FROM wallets
        WHERE roll_no = p_roll_no;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for roll number ' || p_roll_no || '. Skipping transaction.');
            RETURN;
    END;

    -- Check sufficient balance
    IF v_current_balance < p_amount THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Insufficient balance for roll number ' || p_roll_no || '. Skipping transaction.');
        RETURN;
    END IF;

    -- Determine next txn_id
    SELECT NVL(MAX(txn_id), 0) + 1
    INTO v_next_txn_id
    FROM transactions;

    -- Insert transaction record
    INSERT INTO transactions (txn_id, roll_no, amount, txn_time, txn_counter, device_id)
    VALUES (v_next_txn_id, p_roll_no, p_amount, SYSTIMESTAMP,
            (SELECT NVL(MAX(txn_counter), 0) + 1 FROM transactions WHERE roll_no = p_roll_no),
            p_device_id);

    -- Update wallet balance
    UPDATE wallets
    SET balance = v_current_balance - p_amount,
        transaction_counter = transaction_counter + 1,
        last_updated = SYSTIMESTAMP
    WHERE wallet_id = v_wallet_id;

    -- Commit the transaction
    COMMIT;

    -- Fetch the updated balance
    SELECT balance
    INTO v_updated_balance
    FROM wallets
    WHERE roll_no = p_roll_no;

    -- Output the success message with updated balance
    DBMS_OUTPUT.PUT_LINE('Transaction successful for roll number ' || p_roll_no ||
                         ' (new balance: ' || v_updated_balance || ').');

EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error during transaction for roll number ' || p_roll_no || ': ' || SQLERRM);
END;

-- PL/SQL procedure to handle Peer-to-Peer (P2P) payments
CREATE OR REPLACE PROCEDURE process_p2p_payment(
    p_sender_roll_no IN NUMBER,
    p_receiver_roll_no IN NUMBER,
    p_amount IN NUMBER
)
IS
    v_sender_wallet_id NUMBER;
    v_sender_balance NUMBER;
    v_receiver_wallet_id NUMBER;
    v_receiver_balance NUMBER;
    v_next_txn_id NUMBER;
    v_updated_sender_balance NUMBER;
    v_updated_receiver_balance NUMBER;
BEGIN
    -- Validate inputs
    IF p_amount <= 0 THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Payment amount must be positive. Skipping P2P payment.');
        RETURN;
    END IF;
    IF p_sender_roll_no = p_receiver_roll_no THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Sender and receiver cannot be the same. Skipping P2P payment.');
        RETURN;
    END IF;

    -- Check if sender wallet is active
    DECLARE
        v_is_active NUMBER;
    BEGIN
        SELECT is_active
        INTO v_is_active
        FROM wallets
        WHERE roll_no = p_sender_roll_no;
        IF v_is_active = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Warning: Wallet for sender roll number ' || p_sender_roll_no || ' is inactive. Skipping P2P payment.');
            RETURN;
        END IF;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for sender roll number ' || p_sender_roll_no || '. Skipping P2P payment.');
            RETURN;
    END;

    -- Check if receiver wallet is active
    DECLARE
        v_is_active NUMBER;
    BEGIN
        SELECT is_active
        INTO v_is_active
        FROM wallets
        WHERE roll_no = p_receiver_roll_no;
        IF v_is_active = 0 THEN
            DBMS_OUTPUT.PUT_LINE('Warning: Wallet for receiver roll number ' || p_receiver_roll_no || ' is inactive. Skipping P2P payment.');
            RETURN;
        END IF;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for receiver roll number ' || p_receiver_roll_no || '. Skipping P2P payment.');
            RETURN;
    END;

    -- Get sender wallet details
    BEGIN
        SELECT wallet_id, balance
        INTO v_sender_wallet_id, v_sender_balance
        FROM wallets
        WHERE roll_no = p_sender_roll_no;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for sender roll number ' || p_sender_roll_no || '. Skipping P2P payment.');
            RETURN;
    END;

    -- Get receiver wallet details
    BEGIN
        SELECT wallet_id, balance
        INTO v_receiver_wallet_id, v_receiver_balance
        FROM wallets
        WHERE roll_no = p_receiver_roll_no;
    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            DBMS_OUTPUT.PUT_LINE('Warning: No wallet found for receiver roll number ' || p_receiver_roll_no || '. Skipping P2P payment.');
            RETURN;
    END;

    -- Check sufficient balance
    IF v_sender_balance < p_amount THEN
        DBMS_OUTPUT.PUT_LINE('Warning: Insufficient balance for sender roll number ' || p_sender_roll_no || '. Skipping P2P payment.');
        RETURN;
    END IF;

    -- Determine next transaction ID for sender
    SELECT NVL(MAX(txn_id), 0) + 1
    INTO v_next_txn_id
    FROM transactions;

    -- Record transaction for sender (deduction)
    INSERT INTO transactions (txn_id, roll_no, amount, txn_time, txn_counter, device_id)
    VALUES (v_next_txn_id, p_sender_roll_no, p_amount, SYSTIMESTAMP,
            (SELECT NVL(MAX(txn_counter), 0) + 1 FROM transactions WHERE roll_no = p_sender_roll_no),
            'P2P_TRANSFER');

    -- Update sender wallet
    UPDATE wallets
    SET balance = balance - p_amount,
        transaction_counter = transaction_counter + 1,
        last_updated = SYSTIMESTAMP
    WHERE wallet_id = v_sender_wallet_id;

    -- Determine next transaction ID for receiver
    SELECT NVL(MAX(txn_id), 0) + 1
    INTO v_next_txn_id
    FROM transactions;

    -- Record transaction for receiver (addition)
    INSERT INTO transactions (txn_id, roll_no, amount, txn_time, txn_counter, device_id)
    VALUES (v_next_txn_id, p_receiver_roll_no, p_amount, SYSTIMESTAMP,
            (SELECT NVL(MAX(txn_counter), 0) + 1 FROM transactions WHERE roll_no = p_receiver_roll_no),
            'P2P_TRANSFER');

    -- Update receiver wallet
    UPDATE wallets
    SET balance = balance + p_amount,
        transaction_counter = transaction_counter + 1,
        last_updated = SYSTIMESTAMP
    WHERE wallet_id = v_receiver_wallet_id;

    -- Commit the transaction
    COMMIT;

    -- Fetch the updated balances for sender and receiver
    SELECT balance
    INTO v_updated_sender_balance
    FROM wallets
    WHERE roll_no = p_sender_roll_no;

    SELECT balance
    INTO v_updated_receiver_balance
    FROM wallets
    WHERE roll_no = p_receiver_roll_no;

    -- Output the success message with updated balances
    DBMS_OUTPUT.PUT_LINE('P2P payment successful: ' || p_amount || ' transferred from roll number ' || p_sender_roll_no ||
                         ' (new balance: ' || v_updated_sender_balance || ') to roll number ' || p_receiver_roll_no ||
                         ' (new balance: ' || v_updated_receiver_balance || ').');
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        DBMS_OUTPUT.PUT_LINE('Error during P2P payment from roll number ' || p_sender_roll_no || ' to ' ||
                             p_receiver_roll_no || ': ' || SQLERRM);
END;
